def _find(file_key, speaker, fragment):
    for u in TR[file_key]:
        if u["speaker"] == speaker and fragment in u["text"]:
            return True
    return False

def quotes(file_key, items):
    parts = []
    for it in items:
        spk, frag = it[0], it[1]
        interviewer = len(it) > 2 and it[2]
        assert _find(file_key, spk, frag), f"인용 불일치: [{file_key}] {spk}: {frag[:40]}..."
        parts.append(f'(면담자 {spk}) "{frag}"' if interviewer else f'{spk}: "{frag}"')
    return " / ".join(parts)

def content(summary, file_key, items):
    return f"[인터뷰 내용] {summary}\n[발언] {quotes(file_key, items)}"

FONT = Font(name="맑은 고딕", size=10)
HFONT = Font(name="맑은 고딕", size=10, bold=True)
HFILL = PatternFill("solid", fgColor="DDEBF7")
THIN = Side(style="thin", color="999999")
BORDER = Border(left=THIN, right=THIN, top=THIN, bottom=THIN)
ALIGN = Alignment(wrap_text=True, vertical="top")

def est_height(texts, widths):
    lines = 1
    for t, w in zip(texts, widths):
        if not t:
            continue
        cpl = max(int(w / 2.1), 8)
        n = 0
        for ln in str(t).split("\n"):
            n += max(1, math.ceil(len(ln) / cpl))
        lines = max(lines, n)
    return min(409, max(30, lines * 13.5 + 6))

def write_sheet(ws, headers, widths, rows):
    ws.append(headers)
    for c in range(1, len(headers) + 1):
        cell = ws.cell(row=1, column=c)
        cell.font, cell.fill, cell.alignment, cell.border = HFONT, HFILL, ALIGN, BORDER
        ws.column_dimensions[get_column_letter(c)].width = widths[c - 1]
    for r in rows:
        ws.append(list(r))
        i = ws.max_row
        for c in range(1, len(headers) + 1):
            cell = ws.cell(row=i, column=c)
            cell.font, cell.alignment, cell.border = FONT, ALIGN, BORDER
        ws.row_dimensions[i].height = est_height(r, widths)
    ws.freeze_panes = "A2"

def main():
    wb = Workbook()
    ws1 = wb.active
    ws1.title = "법령개정 필요사항"
    h1 = ["연번", "인터뷰 일자", "인터뷰 유형", "대상", "법령명", "내용", "개정 필요사항"]
    w1 = [6, 12, 10, 34, 46, 80, 250]
    rows1 = []
    for i, r in enumerate(ROWS, start=1):
        m = META[r["file"]]
        rows1.append((i, m["date"], m["kind"], m["target"], r["law"], r["content"], r["fix"]))
    write_sheet(ws1, h1, w1, rows1)

    ws2 = wb.create_sheet("검토메모")
    h2 = ["연번", "쟁점", "인터뷰 내용", "법령 확인 결과(근거)", "비고"]
    w2 = [6, 34, 70, 80, 60]
    write_sheet(ws2, h2, w2, [(i, *m) for i, m in enumerate(MEMO, start=1)])

    ws3 = wb.create_sheet("참조 법령·출처")

    w3 = [44, 50, 120]
    write_sheet(ws3, h3, w3, REF)
